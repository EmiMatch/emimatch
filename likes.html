async function swipe(id, action) { if (!me) return; if (action !== "like" && action !== "dislike") { console.error("Acción inválida:", action); return; } const { error } = await db .from("swipes") .upsert( { user_id: me.id, target_user_id: id, action: action }, { onConflict: "user_id,target_user_id" } ); if (error) { console.error(error); notice("No se pudo guardar la acción.", true); return; } if (action === "like") { const { data: other, error: otherError } = await db .from("swipes") .select("action") .eq("user_id", id) .eq("target_user_id", me.id) .eq("action", "like") .maybeSingle(); if (otherError) { console.error(otherError); notice("No se pudo comprobar el Match.", true); return; } if (other?.action === "like") { const [a, b] = [me.id, id].sort(); const { data: existingMatch, error: matchCheckError } = await db .from("matches") .select("id") .eq("user1_id", a) .eq("user2_id", b) .maybeSingle(); if (matchCheckError) { console.error(matchCheckError); notice("No se pudo comprobar el Match.", true); return; } if (!existingMatch) { const { error: matchError } = await db .from("matches") .insert({ user1_id: a, user2_id: b }); if (matchError) { console.error(matchError); notice("No se pudo crear el Match.", true); return; } } notice("💕 ¡Hicieron Match!"); setTimeout(() => { location.href = "conversacion.html?user=" + encodeURIComponent(id); }, 700); return; } notice("❤️ Me gusta enviado."); } else { notice("Perfil descartado."); } const card = document.getElementById("like-" + id); if (card) { card.remove(); } }
