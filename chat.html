async function sendMessage() { const content = input.value.trim(); if (!content || !matchId) return; sendButton.disabled = true; try { /* * Verificar límite Premium antes de enviar. * Gratis: 5 mensajes diarios. * Premium: ilimitados. */ const { data: limitData, error: limitError } = await supabase.rpc("can_send_message"); if (limitError) { console.error( "EmiMatch Premium - límite de mensajes:", limitError ); alert( "No se pudo comprobar el límite de mensajes. Intentá nuevamente." ); return; } if (!limitData?.allowed) { if (limitData?.reason === "not_authenticated") { alert( "Tu sesión expiró. Iniciá sesión nuevamente." ); window.location.href = "index.html"; return; } const remaining = Number(limitData?.remaining ?? 0); alert( "Llegaste al límite de 5 mensajes diarios.\n\n" + "Pasate a EmiMatch Premium para continuar conversando sin límites." ); return; } /* * El límite está permitido. * Se mantiene exactamente el INSERT existente. */ const { data, error } = await supabase .from("messages") .insert({ match_id: matchId, sender_id: currentUser.id, content: content }) .select() .single(); if (error) { console.error(error); alert( "No se pudo enviar el mensaje." ); return; } input.value = ""; if (data) { const empty = messagesBox.querySelector(".empty"); if (empty) { messagesBox.innerHTML = ""; } const div = document.createElement("div"); div.className = "message mine"; div.dataset.messageId = data.id; div.innerHTML = escapeHTML(data.content) + "" + formatTime(data.created_at) + ""; messagesBox.appendChild(div); scrollToBottom(); } input.focus(); } catch (error) { console.error( "EmiMatch Chat:", error ); alert( "No se pudo enviar el mensaje. Intentá nuevamente." ); } finally { sendButton.disabled = false; } }
