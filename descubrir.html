async function swipe(action) { const p = profiles[index]; if (!p) return; if (action !== "like" && action !== "dislike") { console.error("Acción inválida:", action); return; } const r = await client .from("swipes") .upsert( { user_id: user.id, target_user_id: p.id, action: action }, { onConflict: "user_id,target_user_id" } ); if (r.error) { alert("No se pudo guardar la acción: " + r.error.message); return; } if (action === "like") { const a = await client .from("swipes") .select("id") .eq("user_id", p.id) .eq("target_user_id", user.id) .eq("action", "like") .maybeSingle(); if (a.error) { alert("No se pudo comprobar el Match: " + a.error.message); return; } if (a.data) { const ids = [user.id, p.id].sort(); let m = await client .from("matches") .select("id") .eq("user1_id", ids[0]) .eq("user2_id", ids[1]) .maybeSingle(); if (!m.data) { m = await client .from("matches") .insert({ user1_id: ids[0], user2_id: ids[1] }) .select("id") .single(); } if (m.error) { alert( "Like guardado, pero no se pudo crear el Match: " + m.error.message ); return; } showMatch(p); } } index++; render(); }
