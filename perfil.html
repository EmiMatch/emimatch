async function swipe(action, target) { if (busy) return; if (action !== "like" && action !== "dislike") { console.error("Acción inválida:", action); return; } if (!target) { console.error("No se recibió el usuario objetivo."); return; } busy = true; try { /* ========================= GUARDAR SWIPE ========================= */ const { error: swipeError } = await client .from("swipes") .upsert( { user_id: user.id, target_user_id: target, action: action }, { onConflict: "user_id,target_user_id" } ); if (swipeError) { throw swipeError; } /* ========================= COMPROBAR MATCH ========================= */ if (action === "like") { const { data: reverseLike, error: reverseError } = await client .from("swipes") .select("id") .eq("user_id", target) .eq("target_user_id", user.id) .eq("action", "like") .maybeSingle(); if (reverseError) { throw reverseError; } /* ========================= CREAR MATCH ========================= */ if (reverseLike) { const ids = [user.id, target].sort(); const { data: existingMatch, error: matchCheckError } = await client .from("matches") .select("id") .eq("user1_id", ids[0]) .eq("user2_id", ids[1]) .maybeSingle(); if (matchCheckError) { throw matchCheckError; } if (!existingMatch) { const { error: matchInsertError } = await client .from("matches") .insert({ user1_id: ids[0], user2_id: ids[1] }); if (matchInsertError) { throw matchInsertError; } } $("matchBox").classList.add("show"); } } /* ========================= QUITAR PERFIL ACTUAL ========================= */ if (Array.isArray(window.discoveryList)) { window.discoveryList = window.discoveryList.filter( item => item.id !== target ); renderDiscovery( window.discoveryList ); } } catch (error) { console.error( "Error en swipe:", error ); status( "No pudimos registrar tu elección: " + error.message, "error" ); } finally { busy = false; } }
