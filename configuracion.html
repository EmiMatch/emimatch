async function verifyDeleteFunction() { const status = document.getElementById("deleteStatus"); if (!currentUser) { status.textContent = "❌ Sesión no válida."; return; } const firstConfirm = confirm( "⚠️ ¿Estás seguro de que querés eliminar tu cuenta?\n\n" + "Se eliminarán tu perfil, fotos, matches, mensajes, bloqueos, " + "swipes y denuncias relacionadas.\n\n" + "Esta acción no se puede deshacer." ); if (!firstConfirm) return; const secondConfirm = confirm( "🚨 ÚLTIMA CONFIRMACIÓN\n\n" + "Tu cuenta de EmiMatch será eliminada definitivamente.\n\n" + "¿Confirmás la eliminación?" ); if (!secondConfirm) return; status.textContent = "⏳ Eliminando cuenta..."; try { const { data: { session }, error: sessionError } = await supabase.auth.getSession(); if (sessionError || !session) { status.textContent = "❌ Tu sesión expiró. Volvé a iniciar sesión."; return; } const response = await fetch( window.EMIMATCH_CONFIG.supabaseUrl + "/functions/v1/delete-account", { method: "POST", headers: { "Authorization": "Bearer " + session.access_token, "Content-Type": "application/json" }, body: JSON.stringify({ action: "delete" }) } ); let result = {}; try { result = await response.json(); } catch { result = {}; } if (!response.ok || !result.success) { console.error("Error delete-account:", result); status.textContent = "❌ No se pudo eliminar la cuenta: " + (result.error || "Error del servidor."); return; } status.textContent = "✅ Cuenta eliminada correctamente. Cerrando sesión..."; await supabase.auth.signOut(); setTimeout(() => { window.location.href = "index.html"; }, 1500); } catch (error) { console.error(error); status.textContent = "❌ Error de conexión con el servidor."; } }
